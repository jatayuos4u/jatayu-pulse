cmake_minimum_required(VERSION 3.21)
project(JatayuSystemMonitor VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(GNUInstallDirs)
include(CTest)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Widgets Network)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6I18n REQUIRED)
add_library(jatayu_metrics STATIC src/metrics.cpp src/metrics.h src/thermal.cpp src/thermal.h)
target_link_libraries(jatayu_metrics PUBLIC Qt6::Core)
target_include_directories(jatayu_metrics PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
add_executable(jatayu-system-monitor src/main.cpp src/mainwindow.cpp src/mainwindow.h src/chartwidget.cpp src/chartwidget.h assets/pulse.qrc)

target_link_libraries(jatayu-system-monitor PRIVATE jatayu_metrics Qt6::Widgets Qt6::Network KF6::CoreAddons KF6::I18n)
install(TARGETS jatayu-system-monitor RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES assets/jatayu-pulse.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/256x256/apps)
install(FILES assets/jatayu-pulse.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES COPYING DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/jatayu-system-monitor)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/jatayu-system-monitor)
install(FILES packaging/copyright DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/jatayu-system-monitor)
install(FILES packaging/changelog.Debian.gz DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/jatayu-system-monitor)
install(FILES packaging/in.jatayuos.SystemMonitor.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
if(BUILD_TESTING)
 find_package(Qt6 REQUIRED COMPONENTS Test)
 add_executable(test_metrics tests/test_metrics.cpp)
 target_link_libraries(test_metrics PRIVATE jatayu_metrics Qt6::Test)
 add_test(NAME metrics COMMAND test_metrics)
 add_executable(test_thermal tests/test_thermal.cpp)
 target_link_libraries(test_thermal PRIVATE jatayu_metrics Qt6::Test)
 add_test(NAME thermal COMMAND test_thermal)
 add_executable(test_networkrate tests/test_networkrate.cpp)
 target_include_directories(test_networkrate PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
 target_link_libraries(test_networkrate PRIVATE Qt6::Test)
 add_test(NAME networkrate COMMAND test_networkrate)
 add_executable(test_monitorhealth tests/test_monitorhealth.cpp)
 target_include_directories(test_monitorhealth PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
 target_link_libraries(test_monitorhealth PRIVATE Qt6::Test)
 add_test(NAME monitorhealth COMMAND test_monitorhealth)
endif()
